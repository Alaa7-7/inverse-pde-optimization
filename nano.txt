import numpy as np
import matplotlib.pyplot as plt

from src.pde.solver import solve_pde
from src.inverse.optimizer import optimize


# ======================
# Grid setup
# ======================
L = 1.0
Nx = 100
x = np.linspace(0, L, Nx)
dx = x[1] - x[0]

dt = 0.0005
Nt = 200


# ======================
# True parameters
# ======================
v_true = 0.8
D_true = 0.05


# ======================
# Function: run one experiment
# ======================
def run_single_experiment():
    u_obs = solve_pde(v_true, D_true, x, dx, dt, Nt)
    u_obs = u_obs + 0.01 * np.random.randn(len(u_obs))

    v_est, D_est = optimize(u_obs, x, dx, dt, Nt)

    return v_est, D_est


# ======================
# Run multiple experiments
# ======================
runs = 20

v_errors = []
D_errors = []

v_estimates = []
D_estimates = []

for i in range(runs):
    v_est, D_est = run_single_experiment()

    v_estimates.append(v_est)
    D_estimates.append(D_est)

    v_errors.append(abs(v_est - v_true))
    D_errors.append(abs(D_est - D_true))


# ======================
# Print results
# ======================
print("\n===== RESULTS =====")
print("True v:", v_true)
print("True D:", v_true)

print("\nMean v:", np.mean(v_estimates))
print("Mean D:", np.mean(D_estimates))

print("\nMean error v:", np.mean(v_errors))
print("Mean error D:", np.mean(D_errors))


# ======================
# Plot results
# ======================
plt.figure()
plt.bar(["v","D"], [v_true, D_true], alpha=0.6, label="True")
plt.bar(["v","D"], [v_estimates[0], D_estimates[0]], alpha=0.6, label="Estimated")
plt.title("Parameter Estimation")
plt.legend()
plt.show()


plt.figure()
plt.plot(v_errors, label="v error")
plt.plot(D_errors, label="D error")
plt.title("Error across runs")
plt.legend()
plt.show()
